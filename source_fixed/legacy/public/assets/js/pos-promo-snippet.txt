/*
PATCH TAMBAHAN POS.JS UNTUK PROMO

Tempel logic ini di fungsi applyPromo(code), sebelum fetch.
Pastikan dropdown promo otomatis punya name/id salah satu:
- #autoPromoSelect
- [name="promo_id"]
- [data-promo-select]
*/

function getSelectedPromoId() {
    var select = document.querySelector('#autoPromoSelect') ||
        document.querySelector('[name="promo_id"]') ||
        document.querySelector('[data-promo-select]');

    if (!select) return '';
    return select.value || '';
}
